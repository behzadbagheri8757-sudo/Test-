<script>
(function(){
  'use strict';

  // ==== state ====
  const state = {
    count: 1,
    namelen: 'short',
    price: 6,
    mode: 'list',
    lines: [],
    editingId: null,
  };

  const NAMES_SHORT = ['عدس','لوبیا','برنج','نخود','کشمش','زردچوبه','آرد','شکر','نمک','چای'];
  const NAMES_LONG  = [
    'عدس ریز ممتاز ایرانی درجه یک',
    'لوبیا چیتی اعلا بدون سنگ و کاه',
    'برنج طارم هاشمی معطر شمال',
    'نخود نخودی درشت خشکبار باقری',
    'کشمش پلویی سبز تیزاب',
    'زردچوبه هندی آسیاب‌شده',
    'آرد گندم سبوس‌دار نول',
    'شکر سفید تصفیه‌شده',
    'نمک دریاچه ارومیه',
    'چای سیاه قلم شکسته',
  ];

  function faNum(n){
    return String(Math.round(n)).replace(/[0-9]/g, d => '۰۱۲۳۴۵۶۷۸۹'[d])
      .replace(/\B(?=(\d{3})+(?!\d))/g, '٬');
  }
  function faNumFromString(s){
    return String(s).replace(/[0-9]/g, d => '۰۱۲۳۴۵۶۷۸۹'[d]);
  }

  function makeLine(idx){
    const priceBase = state.price === 9
      ? 100000000 + Math.floor(Math.random()*900000000)
      : 150000 + Math.floor(Math.random()*850000);
    const qty = 1 + Math.floor(Math.random()*15);
    const names = state.namelen === 'long' ? NAMES_LONG : NAMES_SHORT;
    return {
      id: 'L' + (Date.now().toString(36)) + Math.random().toString(36).slice(2,5),
      name: names[idx % names.length],
      qty: qty,
      price: priceBase,
      profit: Math.round(priceBase * qty * 0.18),
    };
  }

  function rebuildLines(){
    state.lines = [];
    for (let i = 0; i < state.count; i++) state.lines.push(makeLine(i));
    if (state.mode === 'edit1' && state.lines[0]) state.editingId = state.lines[0].id;
    else state.editingId = null;
  }

  // ==== render ====
  function lineCollapsed(l, idx){
    const total = l.qty * l.price;
    return `
      <div class="line" data-id="${l.id}">
        <div class="line-num">${faNumFromString(String(idx+1))}</div>
        <div class="line-body">
          <div class="line-top">
            <span class="line-name">${escapeHtml(l.name)}</span>
            <span class="line-total">${faNum(total)}<span class="line-unit">ت</span></span>
          </div>
          <div class="line-bottom">
            <span class="line-calc">${faNum(l.qty)} × ${faNum(l.price)}</span>
            <span class="line-profit" data-revealed="0" data-profit-id="${l.id}">
              سود: <span class="profit-value">${faNum(l.profit)}</span>
            </span>
          </div>
        </div>
      </div>`;
  }

  function lineEditing(l, idx){
    const total = l.qty * l.price;
    return `
      <div class="line-editing" data-id="${l.id}">
        <div class="edit-head">
          <span class="edit-index">قلم ${faNumFromString(String(idx+1))}</span>
          <button class="edit-close" data-edit-close="${l.id}" aria-label="بستن">×</button>
        </div>
        <div class="edit-field">
          <label>نام کالا</label>
          <input type="text" value="${escapeHtml(l.name)}" data-field="name" data-id="${l.id}">
        </div>
        <div class="edit-two-col">
          <div class="edit-field">
            <label>تعداد</label>
            <input type="text" inputmode="decimal" value="${faNum(l.qty)}" data-field="qty" data-id="${l.id}">
          </div>
          <div class="edit-field">
            <label>قیمت واحد</label>
            <input type="text" inputmode="decimal" value="${faNum(l.price)}" data-field="price" data-id="${l.id}">
          </div>
        </div>
        <div class="edit-computed">
          <span class="edit-computed-label">جمع</span>
          <span class="edit-computed-value" data-total-for="${l.id}">${faNum(total)} ت</span>
        </div>
        <div class="edit-computed" style="margin-bottom:14px">
          <span class="edit-computed-label">سود این قلم</span>
          <span class="line-profit edit-computed-profit" data-revealed="0" data-profit-id="${l.id}">
            <span class="profit-value">${faNum(l.profit)}</span>
          </span>
        </div>
        <div class="edit-actions">
          <button class="btn-del" data-del="${l.id}">حذف</button>
          <div class="edit-actions-right">
            <button class="btn-ghost" data-cancel="${l.id}">انصراف</button>
            <button class="btn-primary" data-confirm="${l.id}">✓ ثبت قلم</button>
          </div>
        </div>
      </div>`;
  }

  function render(){
    const wrap = document.getElementById('lines');
    wrap.innerHTML = state.lines.map(function(l, i){
      if (state.editingId === l.id) return lineEditing(l, i);
      return lineCollapsed(l, i);
    }).join('');

    const totals = state.lines.reduce(function(acc, l){
      acc.sum += l.qty * l.price;
      acc.profit += l.profit;
      return acc;
    }, {sum: 0, profit: 0});

    const discount = Math.round(totals.sum * 0.03);
    const prevBalance = 4500000;
    const paid = 0;
    const finalTotal = totals.sum - discount;
    const newBalance = prevBalance + finalTotal - paid;

    document.getElementById('summary').innerHTML = `
      <div class="summary-row muted"><span>جمع اقلام</span><b>${faNum(totals.sum)} ت</b></div>
      <div class="summary-row muted"><span>تخفیف</span><b>${faNum(discount)} ت</b></div>
      <div class="summary-row total"><span>جمع فاکتور</span><b>${faNum(finalTotal)} ت</b></div>
      <div class="summary-row muted" style="margin-top:8px"><span>مانده قبلی مشتری</span><b>${faNum(prevBalance)} ت</b></div>
      <div class="summary-row remain"><span>مانده بعد از فاکتور</span><b>${faNum(newBalance)} ت</b></div>
    `;
  }

  function escapeHtml(s){
    return String(s||'').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
  }

  // ==== interactions ====
  document.getElementById('lines').addEventListener('click', function(e){
    // profit reveal
    const profit = e.target.closest('[data-profit-id]');
    if (profit){
      e.stopPropagation();
      profit.setAttribute('data-revealed', profit.getAttribute('data-revealed') === '1' ? '0' : '1');
      return;
    }
    // edit confirm / cancel / delete / close
    const confirmBtn = e.target.closest('[data-confirm]');
    if (confirmBtn){
      state.editingId = null;
      render();
      return;
    }
    const cancelBtn = e.target.closest('[data-cancel], [data-edit-close]');
    if (cancelBtn){
      state.editingId = null;
      render();
      return;
    }
    const delBtn = e.target.closest('[data-del]');
    if (delBtn){
      const id = delBtn.getAttribute('data-del');
      state.lines = state.lines.filter(l => l.id !== id);
      state.editingId = null;
      state.count = state.lines.length;
      render();
      return;
    }
    // collapse row → expand
    const collapsed = e.target.closest('.line[data-id]');
    if (collapsed){
      state.editingId = collapsed.getAttribute('data-id');
      render();
    }
  });

  // live update totals/qty/price while typing
  document.getElementById('lines').addEventListener('input', function(e){
    const inp = e.target.closest('[data-field]');
    if (!inp) return;
    const id = inp.getAttribute('data-id');
    const field = inp.getAttribute('data-field');
    const line = state.lines.find(l => l.id === id);
    if (!line) return;
    if (field === 'name'){
      line.name = inp.value;
    } else if (field === 'qty' || field === 'price'){
      const raw = String(inp.value).replace(/[۰-۹]/g, d => String('۰۱۲۳۴۵۶۷۸۹'.indexOf(d))).replace(/[^\d]/g,'');
      line[field] = Number(raw) || 0;
      const totalEl = document.querySelector('[data-total-for="' + id + '"]');
      if (totalEl) totalEl.textContent = faNum(line.qty * line.price) + ' ت';
      // profit is not live-updated on purpose (Q3 in review):
      // it stays as-is until the line is confirmed and collapsed.
    }
  });

  document.getElementById('add-line').addEventListener('click', function(){
    const l = makeLine(state.lines.length);
    l.name = '';
    l.qty = 1;
    l.price = 0;
    l.profit = 0;
    state.lines.push(l);
    state.editingId = l.id;
    state.count = state.lines.length;
    render();
    // focus first field
    setTimeout(function(){
      const inp = document.querySelector('[data-field="name"][data-id="' + l.id + '"]');
      if (inp) inp.focus();
    }, 50);
  });

  // controls
  document.querySelectorAll('.controls-row').forEach(function(row){
    row.addEventListener('click', function(e){
      const btn = e.target.closest('button[data-count], button[data-namelen], button[data-price], button[data-mode]');
      if (!btn) return;
      row.querySelectorAll('button').forEach(b => b.setAttribute('aria-pressed', b === btn ? 'true' : 'false'));

      if (btn.hasAttribute('data-count')){
        state.count = Number(btn.getAttribute('data-count'));
        state.mode = 'list';
        document.querySelectorAll('[data-mode]').forEach(b => b.setAttribute('aria-pressed', b.getAttribute('data-mode') === 'list' ? 'true' : 'false'));
        rebuildLines();
      } else if (btn.hasAttribute('data-namelen')){
        state.namelen = btn.getAttribute('data-namelen');
        rebuildLines();
      } else if (btn.hasAttribute('data-price')){
        state.price = Number(btn.getAttribute('data-price'));
        rebuildLines();
      } else if (btn.hasAttribute('data-mode')){
        state.mode = btn.getAttribute('data-mode');
        if (state.mode === 'edit1') state.editingId = state.lines[0] ? state.lines[0].id : null;
        else state.editingId = null;
      }
      render();
    });
  });

  // viewport info
  function showWidth(){
    const w = Math.round(window.innerWidth);
    document.getElementById('viewport-info').textContent = 'عرض viewport: ' + w + 'px — ارتفاع: ' + Math.round(window.innerHeight) + 'px';
  }
  window.addEventListener('resize', showWidth);

  // boot
  rebuildLines();
  render();
  showWidth();
